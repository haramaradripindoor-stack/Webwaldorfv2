```python
from groq import Groq
import os

client = Groq(api_key=GROQ_API_KEY)

def traducir_md(contenido, idioma):
    nombre = "inglés" if idioma == "en" else "alemán"
    response = client.chat.completions.create(
        model="llama-3.3-70b-versatile",
        messages=[
            {
                "role": "system",
                "content": f"""Eres un traductor profesional especializado en educación Waldorf.
Traduce el siguiente archivo Markdown al {nombre}.
REGLAS IMPORTANTES:
- Traduce SOLO los valores de texto, NUNCA las claves YAML del frontmatter
- Mantén exactamente la estructura del frontmatter (---, keys, ---)
- Mantén los emojis
- Mantén los links y URLs sin cambiar
- El resultado debe ser solo el archivo traducido, sin explicaciones"""
            },
            {
                "role": "user",
                "content": contenido
            }
        ],
        temperature=0.3,
        max_tokens=2000
    )
    return response.choices[0].message.content

print("✅ Motor de traducción listo")
```

```python
import time

resultados = {}

for nombre_archivo, contenido_bytes in uploaded.items():
    contenido = contenido_bytes.decode('utf-8')
    print(f"\n📄 Procesando: {nombre_archivo}")
    
    for idioma, sufijo in [("en", "en"), ("de", "de")]:
        print(f"   → Traduciendo a {idioma}...")
        try:
            traduccion = traducir_md(contenido, idioma)
            nombre_salida = nombre_archivo.replace(".md", f"-{sufijo}.md")
            
            with open(nombre_salida, 'w', encoding='utf-8') as f:
                f.write(traduccion)
            
            resultados[nombre_salida] = traduccion
            print(f"   ✅ {nombre_salida} generado")
            time.sleep(1)  # respetar rate limit
        except Exception as e:
            print(f"   ❌ Error: {e}")

print(f"\n🎉 Total generados: {len(resultados)} archivos")
```
